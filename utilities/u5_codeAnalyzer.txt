import os
import json
from openai import OpenAI # Keep if actual LLM interaction is planned
from pydantic import BaseModel, Field, ValidationError # Import ValidationError
from tee_logger import start_tee, stop_tee # Used for logging output to file
from typing import List, Dict, Any # For type hinting

# --- Configuration & Global Variables (Consider encapsulating in a config class) ---
# Initialize OpenAI client (remove if not used for actual LLM calls)
# If using Mistral, ensure MISTRAL_API_KEY and MISTRAL_BASE_URL are set in your environment

    llm_client = None # Set to None if initialization fails

# Determine target JSON file path
_CURRENT_FILE_PATH = os.path.abspath(__file__)
_TARGET_JSON_FILE = os.path.splitext(_CURRENT_FILE_PATH)[0] + '.json'

# --- Constants ---
INITIAL_RECORD_STATUS = "Initialized"

# --- Data Model (Pydantic) ---
class CountryRecord(BaseModel): # Renamed for PEP 8 compliance
    """
    Represents a record for country data, including validation rules.
    This model can be used as a schema for tool calling with LLMs.
    """
    country_code: str  = Field(description="Standard Country Code (e.g., 'US', 'GB')", min_length=2, max_length=3)
    country_name: str = Field(description="Full name of the Country (e.g., 'United States', 'United Kingdom')")
    currency:     str = Field(description="Currency code for the country (e.g., 'USD', 'GBP')", min_length=3, max_length=3)
    status:       str = Field(description="State of the record (e.g., 'Initialized', 'Validated', 'Approved')")

# --- Data Repository (Encapsulates persistence logic) ---
class CountryRepository:
    """
    Manages persistence operations for CountryRecord objects to a JSON file.
    """
    def __init__(self, file_path: str):
        self.file_path = file_path
        # Ensure the directory exists
        os.makedirs(os.path.dirname(file_path), exist_ok=True)

    def _read_records(self) -> List[Dict[str, Any]]:
        """Reads all records from the JSON file."""
        if not os.path.exists(self.file_path):
            print(f"Info: JSON file '{self.file_path}' not found. Starting with an empty list.")
            return []
        try:
            with open(self.file_path, 'r', encoding='utf-8') as file:
                return json.load(file)
        except json.JSONDecodeError:
            print(f"Error: JSON file '{self.file_path}' is corrupted or empty. Starting with an empty list.")
            # Optionally, back up the corrupted file before overwriting
            # os.rename(self.file_path, self.file_path + ".corrupted_backup")
            return []
        except IOError as e:
            print(f"Error reading file '{self.file_path}': {e}. Starting with an empty list.")
            return []

    def _write_records(self, records: List[Dict[str, Any]]):
        """Writes all records to the JSON file."""
        try:
            with open(self.file_path, 'w', encoding='utf-8') as file:
                json.dump(records, file, indent=4)
            print(f"Database updated: {len(records)} records saved to '{self.file_path}'.")
        except IOError as e:
            print(f"Critical Error: Could not write to file '{self.file_path}': {e}")

    def add_country_record(self, country_data: CountryRecord) -> Dict[str, Any]:
        """
        Adds a new country record to the database if it doesn't already exist.
        Uses Pydantic model for input validation.
        """
        print(f"Attempting to add record for country - {country_data.country_name}")
        records = self._read_records()
        new_record_dict = country_data.model_dump() # Use model_dump() for Pydantic v2+

        # Check if country already exists by country_code
        existing_record = next((record for record in records
                               if record['country_code'] == new_record_dict['country_code']), None)

        if existing_record:
            print(f"Record already exists for country code '{country_data.country_code}'. No changes made.")
            return {
                "status": "info",
                "message": "Country record already exists",
                "record": existing_record
            }
        else:
            records.append(new_record_dict)
            self._write_records(records)
            print(f"Country record for '{country_data.country_code}' - added to database.")
            return {
                "status": "success",
                "message": "Country record created",
                "record": new_record_dict
            }

# --- Tool Function (Example for LLM integration) ---
def create_country_record_tool(
    country_code: str,
    country_name: str,
    currency: str,
    status: str = INITIAL_RECORD_STATUS
) -> Dict[str, Any]:
    """
    A tool function designed to be called by an LLM agent to create a new country record.
    It validates input using the CountryRecord Pydantic model and then saves it.

    Args:
        country_code (str): Standard Country Code (e.g., 'US').
        country_name (str): Full name of the Country (e.g., 'United States').
        currency (str): Currency code for the country (e.g., 'USD').
        status (str, optional): State of the record. Defaults to "Initialized".

    Returns:
        Dict[str, Any]: A dictionary indicating the status and the created/existing record.
    """
    country_repo = CountryRepository(_TARGET_JSON_FILE)
    try:
        # Validate input using Pydantic model
        new_country = CountryRecord(
            country_code=country_code.strip().upper(),
            country_name=country_name.strip(),
            currency=currency.strip().upper(),
            status=status.strip()
        )
        return country_repo.add_country_record(new_country)
    except ValidationError as e:
        print(f"Validation error when creating country record: {e}")
        return {
            "status": "error",
            "message": f"Invalid input for country record: {e.errors()}",
            "record": None
        }
    except Exception as e:
        print(f"An unexpected error occurred in create_country_record_tool: {e}")
        return {
            "status": "error",
            "message": f"An unexpected error occurred: {str(e)}",
            "record": None
        }

# --- Main Application Logic ---
def main():
    """
    Main function to run the interactive country record creation process.
    """
    # Start tee_logger to capture console output to a file
    log_file_name = f"app_log_{datetime.now().strftime('%Y%m%d_%H%M%S')}.log"

        print("------------------------")

        another = input("\nAdd another country record? (y/N): ").strip().lower()
        if another != 'y':
            break

    # Stop tee_logger
    stop_tee()

if __name__ == '__main__':
